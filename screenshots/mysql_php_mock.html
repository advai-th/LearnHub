// Secure PDO Database Connection function getDatabaseConnection(): PDO { $dsn = "mysql:host=127.0.0.1;dbname=learnhub_db;charset=utf8mb4"; return new PDO($dsn, 'root', '', [ PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION, PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC ]); } // Parameterized Query Execution (Anti-SQLi) $stmt = $pdo->prepare(" SELECT c.*, u.name AS instructor_name, COUNT(l.id) AS total_lessons FROM courses c JOIN users u ON c.instructor_id = u.id LEFT JOIN lessons l ON c.id = l.course_id WHERE c.category = :category GROUP BY c.id "); $stmt->execute([':category' => $category]); $courses = $stmt->fetchAll();
